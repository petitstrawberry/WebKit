cmake_minimum_required(VERSION 3.30)
project(ScarletWebKitDependencies NONE)
option(SCARLET_DEPENDENCIES_CAIRO "Build Cairo and Pixman for the software rendering profile" ON)
include(FetchContent)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_program(SCARLET_HOST_CC NAMES cc clang REQUIRED)
find_program(SCARLET_HOST_CXX NAMES c++ clang++ REQUIRED)
find_program(SCARLET_NATIVE_STRIP NAMES llvm-strip REQUIRED)
if(NOT IS_DIRECTORY "${SCARLET_NATIVE_RUNTIME}/include")
    message(FATAL_ERROR "SCARLET_NATIVE_RUNTIME must be an installed SDK runtime")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/dependencies.lock.json" _lock)
string(JSON _count LENGTH "${_lock}" dependencies)
math(EXPR _last "${_count}-1")
set(_map "{}")
function(_source_url output source)
    string(JSON _url GET "${source}" git_url)
    string(JSON _rev GET "${source}" commit)
    string(REPLACE "https://github.com/" "" _github "${_url}")
    string(REGEX REPLACE "\\.git$" "" _github "${_github}")
    set(${output} "https://codeload.github.com/${_github}/tar.gz/${_rev}" PARENT_SCOPE)
endfunction()
foreach(_index RANGE 0 ${_last})
    string(JSON _item GET "${_lock}" dependencies ${_index})
    string(JSON _name GET "${_item}" name)
    if(NOT SCARLET_DEPENDENCIES_CAIRO AND (_name STREQUAL "cairo" OR _name STREQUAL "pixman"))
        continue()
    endif()
    string(JSON _source GET "${_item}" source)
    string(JSON _kind GET "${_source}" kind)
    if(_kind STREQUAL "git")
        _source_url(_url "${_source}")
        FetchContent_Declare(${_name} URL "${_url}" SOURCE_SUBDIR _scarlet_download_only)
    else()
        string(JSON _url GET "${_source}" url)
        string(JSON _sha GET "${_source}" sha256)
        FetchContent_Declare(${_name} URL "${_url}" URL_HASH "SHA256=${_sha}"
            SOURCE_SUBDIR _scarlet_download_only)
    endif()
    FetchContent_MakeAvailable(${_name})
    string(JSON _subcount ERROR_VARIABLE _suberror LENGTH "${_source}" submodules)
    if(NOT _suberror AND _subcount GREATER 0)
        math(EXPR _sublast "${_subcount}-1")
        foreach(_subindex RANGE 0 ${_sublast})
            string(JSON _sub GET "${_source}" submodules ${_subindex})
            string(JSON _path GET "${_sub}" path)
            _source_url(_url "${_sub}")
            set(_subname "${_name}_${_subindex}")
            FetchContent_Declare(${_subname} URL "${_url}" SOURCE_SUBDIR _scarlet_download_only)
            FetchContent_MakeAvailable(${_subname})
            file(COPY "${${_subname}_SOURCE_DIR}/" DESTINATION "${${_name}_SOURCE_DIR}/${_path}")
        endforeach()
    endif()
    string(JSON _map SET "${_map}" "${_name}" "\"${${_name}_SOURCE_DIR}\"")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/sources.json" "${_map}")
string(JSON _url GET "${_lock}" icu_data url)
string(JSON _sha GET "${_lock}" icu_data sha256)
FetchContent_Declare(icu_payload URL "${_url}" URL_HASH "SHA256=${_sha}"
    DOWNLOAD_NO_EXTRACT TRUE SOURCE_SUBDIR _scarlet_download_only)
FetchContent_MakeAvailable(icu_payload)
file(GLOB _payload "${icu_payload_SOURCE_DIR}/*")
list(LENGTH _payload _payload_count)
if(NOT _payload_count EQUAL 1)
    message(FATAL_ERROR "Expected the single pinned ICU release archive")
endif()
set(_prefix "${CMAKE_CURRENT_BINARY_DIR}/prefix")
set(_archives)
foreach(_name IN ITEMS icuuc icui18n icudata z png16 freetype expat pixman-1 fontconfig
    cairo harfbuzz harfbuzz-icu xml2 sqlite3 jpeg webp webpdemux curl psl ssl crypto)
    if(NOT SCARLET_DEPENDENCIES_CAIRO AND (_name STREQUAL "cairo" OR _name STREQUAL "pixman-1"))
        continue()
    endif()
    list(APPEND _archives "${_prefix}/lib/lib${_name}.a")
endforeach()
set(_profile_args)
if(NOT SCARLET_DEPENDENCIES_CAIRO)
    list(APPEND _profile_args --without-cairo)
endif()
add_custom_command(OUTPUT ${_archives}
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/build.py"
        --sdk "${SCARLET_NATIVE_RUNTIME}" --prefix "${_prefix}"
        --build-dir "${CMAKE_CURRENT_BINARY_DIR}/work"
        --source-map "${CMAKE_CURRENT_BINARY_DIR}/sources.json" --icu-data "${_payload}"
        --cc "${SCARLET_NATIVE_CC}" --cxx "${SCARLET_NATIVE_CXX}"
        --host-cc "${SCARLET_HOST_CC}" --host-cxx "${SCARLET_HOST_CXX}"
        --ar "${SCARLET_NATIVE_AR}" --ranlib "${SCARLET_NATIVE_RANLIB}"
        --strip "${SCARLET_NATIVE_STRIP}" --jobs 4 ${_profile_args}
    DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/build.py"
            "${CMAKE_CURRENT_SOURCE_DIR}/dependencies.lock.json"
    USES_TERMINAL VERBATIM)
add_custom_target(scarlet-dependencies DEPENDS ${_archives})
